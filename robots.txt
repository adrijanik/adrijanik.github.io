// Cookie Consent Management
window.CookieConsent = {
  consentGiven: false,
  preferences: {
    essential: true,
    analytics: false,
    social: false
  },

  init: function() {
    this.loadPreferences();
    if (!this.hasConsent()) {
      this.showBanner();
    } else {
      this.applyConsent();
    }
  },

  hasConsent: function() {
    return localStorage.getItem('cookie-consent') !== null;
  },

  loadPreferences: function() {
    const saved = localStorage.getItem('cookie-preferences');
    if (saved) {
      this.preferences = JSON.parse(saved);
    }
  },

  savePreferences: function() {
    localStorage.setItem('cookie-preferences', JSON.stringify(this.preferences));
  },

  showBanner: function() {
    document.getElementById('cookie-consent').classList.add('show');
  },

  hideBanner: function() {
    document.getElementById('cookie-consent').classList.remove('show');
  },

  acceptAll: function() {
    this.preferences = {
      essential: true,
      analytics: true,
      social: true
    };
    this.giveConsent();
  },

  acceptEssential: function() {
    this.preferences = {
      essential: true,
      analytics: false,
      social: false
    };
    this.giveConsent();
  },

  giveConsent: function() {
    localStorage.setItem('cookie-consent', 'given');
    localStorage.setItem('cookie-consent-date', new Date().toISOString());
    this.savePreferences();
    this.consentGiven = true;
    this.hideBanner();
    this.applyConsent();
  },

  applyConsent: function() {
    // Update checkboxes in settings modal
    document.getElementById('analytics-cookies').checked = this.preferences.analytics;
    document.getElementById('social-cookies').checked = this.preferences.social;

    // Apply consent to scripts
    if (this.preferences.analytics) {
      this.enableGoogleAnalytics();
      this.enableMicrosoftClarity();
    }
    
    if (this.preferences.social) {
      this.enableDisqus();
    }
  },

  enableGoogleAnalytics: function() {
    // Google Analytics will be loaded only after consent
    if (typeof gtag !== 'undefined') {
      // Re-initialize GA with consent
      gtag('consent', 'update', {
        'analytics_storage': 'granted'
      });
    }
  },

  enableMicrosoftClarity: function() {
    // Microsoft Clarity will be loaded only after consent and not on localhost
    if (typeof clarity === 'undefined' && location.hostname !== 'localhost' && location.hostname !== '127.0.0.1') {
      // Load Microsoft Clarity script
      (function(c,l,a,r,i,t,y){
          c[a]=c[a]||function(){(c[a].q=c[a].q||[]).push(arguments)};
          t=l.createElement(r);t.async=1;t.src="https://www.clarity.ms/tag/"+i;
          y=l.getElementsByTagName(r)[0];y.parentNode.insertBefore(t,y);
      })(window, document, "clarity", "script", "w6f8eu0mtq");
    }
  },

  enableDisqus: function() {
    // Disqus will be loaded only after consent
    const disqusElements = document.querySelectorAll('.disqus-wrapper');
    disqusElements.forEach(element => {
      element.style.display = 'block';
    });
  },

  showSettings: function() {
    document.getElementById('cookie-settings-modal').classList.add('show');
  },

  hideSettings: function() {
    document.getElementById('cookie-settings-modal').classList.remove('show');
  },

  saveCustomPreferences: function() {
    this.preferences.analytics = document.getElementById('analytics-cookies').checked;
    this.preferences.social = document.getElementById('social-cookies').checked;
    this.giveConsent();
  },

  withdrawConsent: function() {
    localStorage.removeItem('cookie-consent');
    localStorage.removeItem('cookie-preferences');
    localStorage.removeItem('cookie-consent-date');
    this.consentGiven = false;
    this.preferences = {
      essential: true,
      analytics: false,
      social: false
    };
    this.showBanner();
    
    // Reload page to reset all scripts
    window.location.reload();
  }
};

// Global functions for button onclick handlers
function acceptAllCookies() {
  CookieConsent.acceptAll();
}

function acceptEssentialOnly() {
  CookieConsent.acceptEssential();
}

function showCookieSettings() {
  CookieConsent.showSettings();
}

function hideCookieSettings() {
  CookieConsent.hideSettings();
}

function saveCookiePreferences() {
  CookieConsent.saveCustomPreferences();
}

document.addEventListener('DOMContentLoaded', function() {
  CookieConsent.init();
});

document.addEventListener('keydown', function(e) {
  if (e.key === 'Escape') {
    CookieConsent.hideSettings();
  }
});
